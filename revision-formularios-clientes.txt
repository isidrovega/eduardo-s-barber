async function requestCustomer(action, data) {
    await auth.authStateReady();

    const user = auth.currentUser;

    if (!user) {
        throw new Error(
            "Tu sesión ha expirado. Inicia sesión nuevamente."
        );
    }

    const token = await user.getIdToken();

    let response;

    try {
        response = await fetch(
            "/.netlify/functions/manage-customers",
            {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    Authorization: `Bearer ${token}`
                },
                body: JSON.stringify({
                    action,
                    ...data
                })
            }
        );
    } catch {
        throw new Error(
            "No se pudo conectar con el servidor."
        );
    }

    let result;

    try {
        result = await response.json();
    } catch {
        throw new Error(
            "El servidor devolvió una respuesta no válida."
        );
    }

    if (!response.ok) {
        throw new Error(
            result.error ||
            "No se pudo procesar el cliente."
        );
    }

    return result;
}

export async function findBarberCustomerByPhone(
    phone
) {
    const result = await requestCustomer(
        "find",
        { phone }
    );

    return result.customer || null;
}

export async function saveBarberCustomer(
    name,
    phone
) {
    const result = await requestCustomer(
        "save",
        {
            name,
            phone
        }
    );

    return result.customer;
}

    <script type="module">
        import { auth, db } from "./js/firebase.js";

        import {
            protectPage,
            logout
        } from "./js/auth.js";

        import {
            EmailAuthProvider,
            reauthenticateWithCredential,
            updatePassword
        } from "https://www.gstatic.com/firebasejs/12.4.0/firebase-auth.js";

        import {
            doc,
            getDoc
        } from "https://www.gstatic.com/firebasejs/12.4.0/firebase-firestore.js";

        import { mountBarberCalendar } from "./js/barber-calendar.js";

        import {
    cancelAppointment
} from "./js/appointment-booking.js";

import {
    initializeBarberAppointmentForm
} from "./js/barber-appointment-form.js";

        const loadingSection =
            document.getElementById("loadingSection");

        const panelSection =
            document.getElementById("panelSection");

        const errorSection =
            document.getElementById("errorSection");

        const logoutButton =
            document.getElementById("logoutButton");

        const passwordForm =
            document.getElementById("passwordForm");

        const passwordMessage =
            document.getElementById("passwordMessage");

        const dayNames = {
            monday: "Lunes",
            tuesday: "Martes",
            wednesday: "Miércoles",
            thursday: "Jueves",
            friday: "Viernes",
            saturday: "Sábado",
            sunday: "Domingo"
        };

        function showError(message) {
            loadingSection.hidden = true;
            panelSection.hidden = true;
            errorSection.hidden = false;

            document.getElementById(
                "errorMessage"
            ).textContent = message;
        }

        function renderSchedule(schedule = {}) {
            const container =
                document.getElementById("scheduleList");

            container.replaceChildren();

            for (const [day, label] of Object.entries(dayNames)) {
                const config = schedule[day];

                const row = document.createElement("div");
                row.className = "schedule-row";

                const name = document.createElement("span");
                name.textContent = label;

                const hours = document.createElement("span");

                hours.textContent = config?.enabled
                    ? `${config.start} - ${config.end}`
                    : "Descanso";

                row.append(name, hours);
                container.append(row);
            }
        }

        async function loadPanel() {
            loadingSection.hidden = false;
            panelSection.hidden = true;
            errorSection.hidden = true;

            try {
                const user = await protectPage("barber");

                if (!user) {
                    return;
                }

                const profileSnapshot = await getDoc(
                    doc(db, "users", user.uid)
                );

                if (!profileSnapshot.exists()) {
                    throw new Error(
                        "Tu cuenta no tiene un perfil asignado."
                    );
                }

                const profile = profileSnapshot.data();

                if (
                    profile.role !== "barber" ||
                    profile.active !== true ||
                    !profile.barberId
                ) {
                    throw new Error(
                        "Tu cuenta no tiene acceso activo."
                    );
                }

                const barberSnapshot = await getDoc(
                    doc(db, "barbers", profile.barberId)
                );

                if (!barberSnapshot.exists()) {
                    throw new Error(
                        "No se encontró tu ficha de barbero."
                    );
                }

                const barber = barberSnapshot.data();

                initializeBarberAppointmentForm(barber);

                const calendarContainer =
    document.getElementById("barberCalendar");

if (calendarContainer) {
    mountBarberCalendar(
        calendarContainer,
        profile.barberId
    );
}

                document.getElementById(
                    "welcomeTitle"
                ).textContent = `Bienvenido, ${barber.name || "Barbero"}`;

                document.getElementById(
                    "profileName"
                ).textContent = barber.name || "Sin nombre";

                document.getElementById(
                    "profileEmail"
                ).textContent = user.email || "Sin correo";

                document.getElementById(
                    "profileStatus"
                ).textContent = barber.status === "active"
                    ? "Activo"
                    : "Inactivo";

                renderSchedule(barber.schedule);

                loadingSection.hidden = true;
                panelSection.hidden = false;
                logoutButton.hidden = false;

            } catch (error) {
                console.error(
                    "Error cargando panel de barbero:",
                    error
                );

                showError(
                    "No fue posible consultar tu información. " +
                    "Comprueba tu conexión y tus permisos."
                );
            }
        }

        passwordForm.addEventListener(
            "submit",
            async event => {
                event.preventDefault();

                const user = auth.currentUser;

                if (!user || !user.email) {
                    passwordMessage.textContent =
                        "Debes iniciar sesión nuevamente.";
                    return;
                }

                const currentPassword =
                    document.getElementById(
                        "currentPassword"
                    ).value;

                const newPassword =
                    document.getElementById(
                        "newPassword"
                    ).value;

                const confirmPassword =
                    document.getElementById(
                        "confirmPassword"
                    ).value;

                if (newPassword.length < 8) {
                    passwordMessage.textContent =
                        "Usa al menos 8 caracteres.";
                    return;
                }

                if (newPassword !== confirmPassword) {
                    passwordMessage.textContent =
                        "Las contraseñas nuevas no coinciden.";
                    return;
                }

                const button = document.getElementById(
                    "changePasswordButton"
                );

                button.disabled = true;
                passwordMessage.textContent = "";
                passwordMessage.classList.remove("success");

                try {
                    const credential =
                        EmailAuthProvider.credential(
                            user.email,
                            currentPassword
                        );

                    await reauthenticateWithCredential(
                        user,
                        credential
                    );

                    await updatePassword(
                        user,
                        newPassword
                    );

                    passwordForm.reset();

                    passwordMessage.textContent =
                        "Contraseña actualizada correctamente.";

                    passwordMessage.classList.add("success");

                } catch (error) {
                    console.error(
                        "Error al cambiar contraseña:",
                        error
                    );

                    passwordMessage.textContent =
                        error.code === "auth/invalid-credential"
                            ? "La contraseña actual es incorrecta."
                            : "No se pudo actualizar la contraseña. Intenta nuevamente.";

                } finally {
                    button.disabled = false;
                }
            }
        );

        logoutButton.addEventListener(
            "click",
            async () => {
                logoutButton.disabled = true;

                try {
                    await logout();
                } catch (error) {
                    console.error(error);
                    logoutButton.disabled = false;
                }
            }
        );

        document.getElementById(
            "retryButton"
        ).addEventListener("click", loadPanel);

        loadPanel();


const barberCalendarElement =
    document.getElementById("barberCalendar");

if (barberCalendarElement) {
    barberCalendarElement.addEventListener(
        "appointment:cancel",
        async event => {
            const appointment = event.detail?.appointment;

            if (!appointment?.id) {
                alert("No se pudo identificar la cita.");
                return;
            }

            const confirmed = window.confirm(
                `¿Cancelar la cita de ${appointment.clientName} ` +
                `del ${appointment.date} a las ${appointment.time}?`
            );

            if (!confirmed) {
                return;
            }

            try {
                await cancelAppointment(appointment.id);

                alert("La cita se canceló correctamente.");
            } catch (error) {
                console.error(
                    "Error al cancelar la cita:",
                    error
                );

                alert(
                    error.message ||
                    "No se pudo cancelar la cita."
                );
            }
        }
    );
}
